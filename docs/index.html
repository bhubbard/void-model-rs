void-model-rs — Physics-Aware Video Object, Shadow & Reflection Removal in Rust

void-model-rs is a pure Rust port of Netflix's VOID model that removes unwanted objects from video along with their cast shadows, reflections, and physical scene interactions. It combines quad-mask counterfactual causal reasoning with spatiotemporal diffusion inpainting to heal occluded backgrounds seamlessly.

          04 // WGPU Pass 2 Latent Noise Advection Shader (WGSL)
          src/flow/warped_noise.rs
        

        
@compute @workgroup_size(8, 8, 1)
fn main(@builtin(global_invocation_id) gid: vec3<u32>) {
    let x = gid.x; let y = gid.y;
    if (x >= params.width || y >= params.height) { return; }
    let uv = flow_uv[y * params.width + x];
    let sx = clamp(f32(x) - uv.x, 0.0, f32(params.width - 1u));
    let sy = clamp(f32(y) - uv.y, 0.0, f32(params.height - 1u));
    let x0 = u32(floor(sx)); let y0 = u32(floor(sy));
    let x1 = min(x0 + 1u, params.width - 1u);
    let y1 = min(y0 + 1u, params.height - 1u);
    let dx = sx - f32(x0); let dy = sy - f32(y0);
    let w_warp = sqrt(max(1.0 - params.lambda * params.lambda, 0.0));
    let plane = params.height * params.width;
    for (var c: u32 = 0u; c < params.channels; c = c + 1u) {
        let base = c * plane;
        let warped = (1.0 - dx) * (1.0 - dy) * prev_latent[base + y0 * params.width + x0]
                   + dx * (1.0 - dy) * prev_latent[base + y0 * params.width + x1]
                   + (1.0 - dx) * dy * prev_latent[base + y1 * params.width + x0]
                   + dx * dy * prev_latent[base + y1 * params.width + x1];
        let idx = base + y * params.width + x;
        out_latent[idx] = w_warp * warped + params.lambda * fresh_latent[idx];
    }
}
